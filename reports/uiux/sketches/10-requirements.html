the subset, stated

  

    

      
Rendered. ATX headings (demoted — the page owns its
      <h1>, a quoted document may not create a second one), paragraphs,
      fenced and indented code, unordered and ordered lists to depth 3, block quotes to depth 2,
      thematic breaks, GFM pipe tables with real <th scope>, inline code,
      strong, emphasis, strikethrough, and links under the policy below. Two HTML tag names are
      recognised as block markers rather than parsed as HTML —
      <details> and <summary> — because this
      README's structure depends on them and dropping them would hide content. Two line-anchored
      literals are not an HTML parser.

    

    

      
Refused, and shown refusing. All other raw HTML, block and inline.
      Images (img-src 'self' data: means a corpus image is either broken or
      a payload). Footnotes, definition lists, math, heading-id attributes, and anything else the
      scanner does not recognise — which falls through to paragraph text, never to nothing.
      Link targets survive in exactly three shapes: an in-document
      #anchor; a relative repo path matching
      ^[A-Za-z0-9._/-]+$ with no .. segment, turned into
      an in-app route by the router and never by assigning href from data;
      and http(s)://, which renders as text with the full URL visible and
      is not a link. Everything else is literal text.

    

  

  

    Size, measured rather than guessed: the renderer running on this page —
    blocks(), inline(), build()
    and the link policy — is 324 lines of code, 411 with its comments, and it renders
    the whole subset above including the hostile cases. Shipped it needs hardened inline edge cases
    and section-range rendering for R2, so budget ~450–500 renderer lines + ~120 of viewer
    chrome + ~250 of tests. marked is ~4k lines and renders raw HTML by
    default, which would then need a sanitiser — a second dependency. The subset is smaller
    because it refuses.

function paint(){
  const d = DOCS[current];
  $('src').textContent = d.src;
  $('dh-path').textContent = d.path;
  $('dh-meta').textContent = d.meta;
  $('dh-origin').textContent = d.origin;

  const out = $('out');
  out.replaceChildren();                       // never innerHTML = ''
  out.setAttribute('dir', current === 'hebrew' ? 'rtl' : 'ltr');
  out.setAttribute('lang', current === 'hebrew' ? 'he' : 'en');

  if (showingSource){
    const pre = document.createElement('pre');
    pre.className = 'srcpane';
    pre.textContent = d.src;
    out.append(pre);
  } else {
    const {node, refusals} = renderMarkdown(d.src);
    out.append(node);
    paintLedger(refusals);
    return;
  }
  paintLedger(renderMarkdown(d.src).refusals);
}

function paintLedger(refusals){
  const box = $('ledger');
  box.replaceChildren();
  box.classList.toggle('clean', refusals.length === 0);
  const p = document.createElement('p');
  p.style.margin = '0';
  p.className = 'eyebrow';
  p.textContent = refusals.length === 0
    ? 'nothing on this page was refused — every construct in it is rendered'
    : refusals.length + ' thing(s) on this page are shown as text rather than rendered';
  box.append(p);
  if (refusals.length){
    const ul = document.createElement('ul');
    const seen = new Map();
    for (const r of refusals){
      const key = r.kind;
      if (!seen.has(key)) seen.set(key, []);
      seen.get(key).push(r.line);
    }
    for (const [kind, lines] of seen){
      const li = document.createElement('li');
      const b = document.createElement('bdi');
      b.textContent = kind;
      li.append(b, document.createTextNode(' — line' + (lines.length>1?'s':'') + ' ' + lines.join(', ')));
      ul.append(li);
    }
    box.append(ul);
    const n = document.createElement('p');
    n.className = 'note';
    n.style.marginBottom = '0';
    n.textContent = 'Each is kept, marked and counted. Nothing is dropped silently — and the whole '
      + 'source is one button away, so the refusal costs the reader information about form, never content.';
    box.append(n);
  }
}

function buildHelp(region, payload){
  region.replaceChildren();
  const sec = (title, srcline, node) => {
    const s = document.createElement('section');
    const h = document.createElement('h4'); h.textContent = title;
    s.append(h, node);
    if (srcline){ const c = document.createElement('p'); c.className = 'src'; c.textContent = srcline; s.append(c); }
    return s;
  };
  const md = (text) => { const {node} = renderMarkdown(text); const d = document.createElement('div'); d.append(node); return d; };

  region.append(sec('what it is', payload.what.src, md(payload.what.md)));
  region.append(sec('in this project', payload.here.src, md(payload.here.md)));

  const cmdBox = document.createElement('div');
  cmdBox.className = 'cmd flat';
  cmdBox.style.marginBottom = '0';
  const code = document.createElement('code');
  const idx = payload.cmd.indexOf('--yes');
  if (idx >= 0){
    code.append(document.createTextNode(payload.cmd.slice(0, idx)));
    const y = document.createElement('span'); y.className = 'yes'; y.textContent = '--yes';
    code.append(y, document.createTextNode(payload.cmd.slice(idx + 5)));
  } else code.textContent = payload.cmd;
  cmdBox.append(code);
  region.append(sec('run this in your terminal — this page never writes', null, cmdBox));

  const land = document.createElement('p'); land.textContent = payload.landing;
  region.append(sec('how you will know it worked', 'the command definition\'s landing predicate', land));

  const ul = document.createElement('ul');
  ul.style.margin = '0'; ul.style.paddingInlineStart = '20px';
  for (const [label, href] of payload.more){
    const li = document.createElement('li');
    const a = document.createElement('a'); a.href = href; a.append(txt(label));
    const u = document.createElement('span'); u.className = 'ext'; u.textContent = ' ⟨' + href + '⟩';
    li.append(a, u); ul.append(li);
  }
  region.append(sec('read more — deep-linked to the exact section', null, ul));
}

/* ── R5: measure the gloss, in this browser, from the resolved tokens ────── */
function srgbToLin(c){ c /= 255; return c <= 0.03928 ? c/12.92 : Math.pow((c+0.055)/1.055, 2.4); }
function lum(rgb){ return 0.2126*srgbToLin(rgb[0]) + 0.7152*srgbToLin(rgb[1]) + 0.0722*srgbToLin(rgb[2]); }
function ratio(a, b){ const la = lum(a), lb = lum(b); const hi = Math.max(la,lb), lo = Math.min(la,lb);
  return (hi + 0.05) / (lo + 0.05); }
function parseRGB(s){
  const m = /rgba?\(\s*([\d.]+)[,\s]+([\d.]+)[,\s]+([\d.]+)(?:[,/\s]+([\d.]+))?/.exec(s);
  return m ? [+m[1], +m[2], +m[3], m[4] === undefined ? 1 : +m[4]] : [0,0,0,1];
}
function over(fg, bg){ const a = fg[3];
  return [fg[0]*a + bg[0]*(1-a), fg[1]*a + bg[1]*(1-a), fg[2]*a + bg[2]*(1-a), 1]; }

function measure(scopeEl, tableEl){
  const probe = document.createElement('div');
  scopeEl.append(probe);
  const read = (v) => { probe.style.color = 'var(' + v + ')';
    return parseRGB(getComputedStyle(probe).color); };
  const panel = read('--panel');
  const ink   = read('--ink');
  const dim   = read('--dim');
  const faint = read('--faint');
  const top   = read('--sheen-top');
  const bot   = read('--sheen-bottom');
  probe.remove();

  const sheenTop = over(top, panel);      // worst case at the top of the card
  const sheenBot = over(bot, panel);      // worst case at the bottom
  const rows = [
    ['--ink on the card at its lightest',  ratio(ink, sheenTop), 4.5],
    ['--ink on the card at its darkest',   ratio(ink, sheenBot), 4.5],
    ['--dim on the card at its lightest',  ratio(dim, sheenTop), 4.5],
    ['--dim on the card at its darkest',   ratio(dim, sheenBot), 4.5],
    ['--faint (corrected) worst point',    Math.min(ratio(faint, sheenTop), ratio(faint, sheenBot)), 4.5],
    ['--faint on flat --panel, for comparison', ratio(faint, panel), 4.5],
  ];

  tableEl.replaceChildren();
  const cap = document.createElement('p');
  const thead = document.createElement('thead');
  const htr = document.createElement('tr');
  for (const h of ['measured on the glossed card','ratio','min','']){
    const th = document.createElement('th'); th.setAttribute('scope','col'); th.textContent = h; htr.append(th);
  }
  thead.append(htr);
  const tbody = document.createElement('tbody');
  for (const [label, r, min] of rows){
    const tr = document.createElement('tr');
    const a = document.createElement('td'); a.textContent = label;
    const b = document.createElement('td'); b.className = 'num'; b.textContent = r.toFixed(2) + ':1';
    const c = document.createElement('td'); c.className = 'num'; c.textContent = min.toFixed(1);
    const d = document.createElement('td');
    d.className = r >= min ? 'pass' : 'fail';
    d.textContent = r >= min ? 'pass' : 'FAIL';
    tr.append(a,b,c,d); tbody.append(tr);
  }
  tableEl.append(thead, tbody);
}
